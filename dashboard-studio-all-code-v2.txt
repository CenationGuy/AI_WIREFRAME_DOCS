// POST /api/chat  { messages:[{role,content}] }  ->  { reply:"..." }
async function askAssistant(messages) {
  const res = await fetch(CHAT_ENDPOINT, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ messages }),
  });
  if (!res.ok) throw new Error("bad status");
  const data = await res.json();
  return (data.reply || data.content || data.message || "").trim();
}

function renderRich(text, isUser) {
  return text.split("\n").map((line, i) => {
    if (line.trim() === "") return <div key={i} className="h-1.5" />;
    const bullet = /^\s*[-*]\s+/.test(line);
    const clean = line.replace(/^\s*[-*]\s+/, "");
    const parts = clean.split(/(\*\*[^*]+\*\*|`[^`]+`)/g).filter(Boolean).map((p, j) => {
      if (p.startsWith("**") && p.endsWith("**")) return <strong key={j} className="font-bold">{p.slice(2, -2)}</strong>;
      if (p.startsWith("`") && p.endsWith("`")) return <code key={j} className={"rounded px-1.5 py-0.5 font-mono text-[12px] " + (isUser ? "bg-white/20" : "bg-slate-200/70")}>{p.slice(1, -1)}</code>;
      return <span key={j}>{p}</span>;
    });
    return <div key={i} className="flex gap-1.5">{bullet && <span className={isUser ? "text-rose-200" : "text-[#E60000]"}>•</span>}<span>{parts}</span></div>;
  });
}

  const openChat = () => { setOpen(true); setNudge(false); setNudgeDone(true); };
  const dismissNudge = () => { setNudge(false); setNudgeDone(true); };
  const grow = () => { const el = taRef.current; if (!el) return; el.style.height = "auto"; el.style.height = Math.min(el.scrollHeight, 96) + "px"; };

  const send = async (text) => {
    const content = (text ?? input).trim();
    if (!content || loading) return;
    const next = [...messages, { role: "user", content }];
    setMessages(next); setInput(""); setError("");
    if (taRef.current) taRef.current.style.height = "auto";
    setLoading(true);
    try {
      const reply = await askAssistant(next);
      setMessages((m) => [...m, { role: "assistant", content: reply || "I'm not sure — try rephrasing." }]);
    } catch {
      setError("Couldn't reach the assistant. Please try again.");
    } finally { setLoading(false); }
  };

  const onKey = (e) => { if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); send(); } };

  if (!open) {
    return (
      <>
        {nudge && !nudgeDone && (
          <div className="vf-nudge fixed bottom-24 right-6 z-[59] w-[252px] rounded-2xl rounded-br-md border border-slate-200 bg-white p-3.5 pr-9 shadow-[0_20px_44px_-16px_rgba(15,23,42,0.36)]">
            <button onClick={dismissNudge} className="absolute right-2 top-2 grid h-6 w-6 place-items-center rounded-md text-slate-400 hover:bg-slate-100"><X size={13} /></button>
            <div className="flex items-center gap-1.5 text-[13px] font-bold text-slate-800"><Sparkles size={14} className="text-[#E60000]" /> Studio Assistant</div>
            <div className="mt-1 text-xs leading-relaxed text-slate-500">Need a hand? I can suggest KPIs, chart types and layouts for your dashboard.</div>
          </div>
        )}
        <button onClick={openChat} aria-label="Open Studio Assistant"
          className="vf-red-grad fixed bottom-6 right-6 z-[60] flex h-14 items-center gap-2.5 rounded-[30px] pl-3.5 pr-5 font-bold text-white shadow-[0_16px_34px_-10px_rgba(230,0,0,0.6)] transition hover:-translate-y-0.5">
          <span className="vf-ping pointer-events-none absolute inset-0 rounded-[30px] border-2 border-[#E60000]" />
          <span className="grid h-8 w-8 place-items-center rounded-full bg-white/20"><MessageCircle size={18} /></span>
          <span className="hidden sm:inline">Ask Studio Assistant</span>
        </button>
      </>
    );
  }

  return (
    <div className="vf-slide fixed bottom-6 right-6 z-[61] flex h-[min(624px,82vh)] w-[392px] max-w-[calc(100vw-32px)] flex-col overflow-hidden rounded-[20px] border border-slate-200 bg-white shadow-[0_32px_70px_-22px_rgba(15,23,42,0.42)]">
      <div className="flex items-center gap-3 border-b border-slate-200 bg-slate-50 px-4 py-3.5">
        <div className="vf-red-grad relative grid h-9 w-9 place-items-center rounded-xl text-white">
          <Bot size={19} /><span className="absolute -bottom-0.5 -right-0.5 h-3 w-3 rounded-full border-2 border-white bg-emerald-500" />
        </div>
        <div>
          <div className="text-sm font-bold leading-tight text-slate-800">Studio Assistant</div>
          <div className="mt-0.5 flex items-center gap-1.5 text-[11.5px] text-slate-500"><span className="h-1.5 w-1.5 rounded-full bg-emerald-500" /> Online</div>
        </div>
        <div className="ml-auto flex gap-1">
          {messages.length > 0 && (
            <button onClick={() => { setMessages([]); setError(""); }} className="grid h-[30px] w-[30px] place-items-center rounded-lg text-slate-500 hover:bg-slate-200/60 hover:text-slate-800"><RefreshCw size={15} /></button>
          )}
          <button onClick={() => setOpen(false)} className="grid h-[30px] w-[30px] place-items-center rounded-lg text-slate-500 hover:bg-slate-200/60 hover:text-slate-800"><ChevronDown size={17} /></button>
        </div>
      </div>

      <div className="flex-1 space-y-3.5 overflow-y-auto p-4">
        {messages.length === 0 && !loading && (
          <div>
            <div className="pt-3 text-center">
              <div className="mx-auto grid h-11 w-11 place-items-center rounded-2xl border border-rose-200 bg-rose-50 text-[#E60000]"><Sparkles size={20} /></div>
              <h4 className="mt-3 text-[15px] font-bold text-slate-800">How can I help?</h4>
              <p className="mx-auto mt-1 max-w-[30ch] text-[12.5px] leading-relaxed text-slate-500">Ask about KPIs, chart types or how to tailor a dashboard to your audience.</p>
            </div>
            <div className="mt-4 flex flex-col gap-2">
              {SUGGESTIONS.map((s) => (
                <button key={s} onClick={() => send(s)} className="flex items-center gap-2 rounded-xl border border-slate-200 bg-slate-50 px-3 py-2.5 text-left text-[12.5px] text-slate-700 transition hover:border-[#E60000] hover:bg-rose-50/40 hover:text-[#E60000]">
                  <ChevronRight size={14} className="shrink-0 opacity-60" /> {s}
                </button>
              ))}
            </div>
          </div>
        )}

        {messages.map((m, i) => {
          const isUser = m.role === "user";
          return (
            <div key={i} className={"flex items-end gap-2.5 " + (isUser ? "flex-row-reverse" : "")}>
              {!isUser && <span className="vf-red-grad grid h-7 w-7 shrink-0 place-items-center rounded-lg text-white"><Bot size={15} /></span>}
              <div className={"max-w-[82%] break-words rounded-2xl px-3.5 py-2.5 text-[13px] leading-relaxed " + (isUser ? "vf-red-grad rounded-br-md text-white shadow-[0_8px_18px_-10px_rgba(230,0,0,0.6)]" : "rounded-bl-md border border-slate-200 bg-slate-50 text-slate-800")}>
                {renderRich(m.content, isUser)}
              </div>
            </div>
          );
        })}

        {loading && (
          <div className="flex items-end gap-2.5">
            <span className="vf-red-grad grid h-7 w-7 shrink-0 place-items-center rounded-lg text-white"><Bot size={15} /></span>
            <div className="rounded-2xl rounded-bl-md border border-slate-200 bg-slate-50 px-4 py-3">
              <div className="flex gap-1">
                <span className="vf-dot h-1.5 w-1.5 rounded-full bg-slate-300" />
                <span className="vf-dot h-1.5 w-1.5 rounded-full bg-slate-300" style={{ animationDelay: "0.18s" }} />
                <span className="vf-dot h-1.5 w-1.5 rounded-full bg-slate-300" style={{ animationDelay: "0.36s" }} />
              </div>
            </div>
          </div>
        )}

        {error && <div className="rounded-xl border border-rose-200 bg-rose-50 px-3 py-2 text-center text-xs text-rose-700">{error}</div>}
        <div ref={endRef} />
      </div>

      <div className="border-t border-slate-200 bg-slate-50 p-3">
        <div className="flex items-end gap-2 rounded-2xl border border-slate-200 bg-white py-1.5 pl-3 pr-1.5 focus-within:border-[#E60000] focus-within:ring-[3px] focus-within:ring-[#ffe4e4]">
          <textarea ref={taRef} rows={1} value={input} onChange={(e) => { setInput(e.target.value); grow(); }} onKeyDown={onKey}
            placeholder="Ask about your dashboard…" className="max-h-24 flex-1 resize-none bg-transparent py-1.5 text-[13px] leading-relaxed text-slate-800 outline-none placeholder:text-slate-400" />
          <button onClick={() => send()} disabled={!input.trim() || loading}
            className="vf-red-grad grid h-9 w-9 shrink-0 place-items-center rounded-xl text-white transition hover:-translate-y-0.5 disabled:translate-y-0 disabled:opacity-40"><Send size={16} /></button>
        </div>
        <div className="mt-2 text-center text-[10.5px] text-slate-400">Enter to send · Shift + Enter for a new line</div>
      </div>
    </div>
  );
}

async function generateDashboard({ file, persona, tool, instructions }) {
  if (!file) {
    throw new Error("Please upload a CSV file first.");
  }

  const formData = new FormData();
  formData.append("file", file);

  const response = await fetch("/api/generate-dashboard", {
    method: "POST",
    body: formData,
  });

  if (!response.ok) {
    let errorMessage = "Dashboard generation failed.";
    try {
      const errorData = await response.json();
      errorMessage = errorData.detail || errorMessage;
    } catch {
      // Keep default error message
    }
    throw new Error(errorMessage);
  }

  const data = await response.json();
  return data;
}

export default function App() {
  const [sessions, setSessions] = useState(INITIAL_SESSIONS);
  const [activeId, setActiveId] = useState(INITIAL_SESSIONS[0].id);
  const [query, setQuery] = useState("");

  const [persona, setPersona] = useState("Executive");
  const [tool, setTool] = useState("power_bi");
  const [file, setFile] = useState(null);
  const [instructions, setInstructions] = useState("");

  const [phase, setPhase] = useState("idle");
  const [result, setResult] = useState(null);
  const [error, setError] = useState("");

  const active = sessions.find((s) => s.id === activeId) || null;

  const handleNew = () => {
    const id = Date.now();
    setSessions((prev) => [
      { id, title: "Untitled dashboard", market: "DE", when: "Today" },
      ...prev,
    ]);
    setActiveId(id);
    setFile(null);
    setInstructions("");
    setPhase("idle");
    setResult(null);
    setError("");
  };

  const handleGenerate = async () => {
    setPhase("working");
    setError("");
    setResult(null);

    try {
      const response = await generateDashboard({ file, persona, tool, instructions });
      console.log("Backend response:", response);
      setResult(response);

      // Update the session title with the generated dashboard title
      if (response.dashboard_spec?.dashboard_title) {
        setSessions((prev) =>
          prev.map((s) =>
            s.id === activeId
              ? { ...s, title: response.dashboard_spec.dashboard_title }
              : s
          )
        );
      }

      setPhase("done");
    } catch (err) {
      console.error("Dashboard generation error:", err);
      setError(err.message || "Something went wrong.");
      setPhase("error");
    }
  };

  return (
    <div className="flex h-screen w-full bg-[#f5f7fa] text-slate-800 antialiased">
      <Sidebar
        sessions={sessions}
        activeId={activeId}
        query={query}
        onQuery={setQuery}
        onSelect={setActiveId}
        onNew={handleNew}
      />

      <main className="flex min-w-0 flex-1 flex-col">
        <TopBar title={active ? active.title : "New dashboard"} />

        <div className="flex-1 overflow-y-auto p-6">
          <div className="mx-auto max-w-[1180px]">
            <div className="mb-5">
              <h1 className="text-[23px] font-extrabold tracking-tight text-slate-900">Create a dashboard</h1>
              <p className="mt-1.5 max-w-2xl text-[14px] text-slate-500">
                Upload a CSV file and generate an AI-powered dashboard with one or more sheets.
              </p>
            </div>

            <div className="grid grid-cols-1 items-start gap-5 lg:grid-cols-12">
              <div className="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm lg:col-span-5">
                <GeneratorForm
                  persona={persona} setPersona={setPersona}
                  tool={tool} setTool={setTool}
                  file={file} setFile={setFile}
                  instructions={instructions} setInstructions={setInstructions}
                  phase={phase} onGenerate={handleGenerate}
                />
              </div>

              <div className="overflow-hidden rounded-2xl border border-slate-200 bg-white shadow-[0_18px_44px_-26px_rgba(15,23,42,0.28)] lg:col-span-7">
                <PreviewPanel phase={phase} result={result} error={error} tool={tool} persona={persona} />
              </div>
            </div>
          </div>
        </div>
      </main>

      <ChatAssistant />
    </div>
  );
}
