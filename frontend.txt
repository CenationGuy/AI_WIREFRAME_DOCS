generatorform.jsx

import { useRef } from "react";
import { UploadCloud, Sparkles, FileSpreadsheet, X } from "lucide-react";
import Step from "./Step.jsx";

const PERSONAS = ["Executive", "Analyst", "Operations", "Finance"];
const TOOLS = [
  { id: "power_bi", name: "Power BI", vendor: "Microsoft", dot: "bg-amber-400" },
  { id: "qlik", name: "Qlik Sense", vendor: "Qlik", dot: "bg-emerald-500" },
  { id: "sac", name: "SAP Analytics", vendor: "SAC", dot: "bg-sky-500" },
];

export default function GeneratorForm({
  persona, setPersona, tool, setTool, file, setFile,
  instructions, setInstructions, phase, onGenerate,
}) {
  const fileRef = useRef(null);
  const onPick = (e) => { if (e.target.files[0]) setFile(e.target.files[0]); };

  return (
    <div className="flex flex-col">
      {/* 1 - Upload */}
      <Step n="1" label="Dataset" hint="CSV up to 50 MB">
        {!file ? (
          <button
            type="button"
            onClick={() => fileRef.current && fileRef.current.click()}
            className="group flex w-full items-center gap-2.5 rounded-lg border border-dashed border-slate-200 bg-slate-50/60 p-2.5 text-left transition-all hover:border-[#E60000] hover:bg-rose-50/20"
          >
            <div className="grid h-8 w-8 shrink-0 place-items-center rounded-lg bg-white text-[#E60000] shadow-2xs">
              <UploadCloud size={15} />
            </div>
            <div className="min-w-0 flex-1">
              <div className="text-xs font-semibold text-slate-800">Choose CSV file</div>
              <div className="truncate text-[10px] text-slate-400">Schema profiled automatically</div>
            </div>
          </button>
        ) : (
          <div className="flex items-center justify-between rounded-lg border border-emerald-200 bg-emerald-50/60 px-3 py-2 text-emerald-900">
            <div className="flex items-center gap-2 truncate">
              <FileSpreadsheet size={15} className="shrink-0 text-emerald-600" />
              <span className="truncate font-mono text-xs font-medium">{file.name}</span>
            </div>
            <button
              onClick={() => setFile(null)}
              className="rounded p-0.5 text-slate-400 hover:text-slate-700"
            >
              <X size={13} />
            </button>
          </div>
        )}
        <input ref={fileRef} type="file" accept=".csv" hidden onChange={onPick} />
      </Step>

      {/* 2 - Persona */}
      <Step n="2" label="Target Persona" hint="Audience">
        <div className="grid grid-cols-4 gap-1.5">
          {PERSONAS.map((p) => {
            const on = p === persona;
            return (
              <button
                key={p}
                type="button"
                onClick={() => setPersona(p)}
                className={
                  "rounded-lg py-1.5 text-center text-xs font-medium transition-all " +
                  (on
                    ? "bg-[#E60000] font-semibold text-white shadow-2xs"
                    : "border border-slate-200/80 bg-slate-50/50 text-slate-600 hover:bg-white")
                }
              >
                {p}
              </button>
            );
          })}
        </div>
      </Step>

      {/* 3 - Target BI Tool */}
      <Step n="3" label="Target BI Tool" hint="Visual style">
        <div className="grid grid-cols-3 gap-1.5">
          {TOOLS.map((t) => {
            const on = t.id === tool;
            return (
              <button
                key={t.id}
                type="button"
                onClick={() => setTool(t.id)}
                className={
                  "rounded-lg border p-2 text-left transition-all " +
                  (on
                    ? "border-[#E60000] bg-rose-50/30 ring-1 ring-[#E60000]"
                    : "border-slate-200/80 bg-slate-50/30 hover:bg-white")
                }
              >
                <div className="flex items-center gap-1.5">
                  <span className={"h-1.5 w-1.5 rounded-full " + t.dot} />
                  <span className="text-xs font-semibold text-slate-800">{t.name}</span>
                </div>
                <div className="mt-0.5 text-[9px] text-slate-400">{t.vendor}</div>
              </button>
            );
          })}
        </div>
      </Step>

      {/* 4 - Instructions */}
      <Step n="4" label="Prompt & Instructions" hint="Plain text">
        <textarea
          rows={2}
          value={instructions}
          onChange={(e) => setInstructions(e.target.value)}
          placeholder="e.g. Total revenue KPI, regional bar chart, top 5 products, churn trend."
          className="w-full resize-none rounded-lg border border-slate-200/80 bg-slate-50/40 p-2.5 text-xs leading-relaxed text-slate-700 outline-none placeholder:text-slate-400 focus:border-[#E60000] focus:bg-white"
        />
      </Step>

      {/* Action Button */}
      <div className="mt-3.5">
        <button
          type="button"
          onClick={onGenerate}
          disabled={phase === "working"}
          className="flex w-full items-center justify-center gap-1.5 rounded-lg bg-[#E60000] py-2.5 text-xs font-semibold text-white shadow-2xs transition-all hover:bg-[#C20000] active:scale-[0.99] disabled:opacity-60"
        >
          <Sparkles size={14} />
          {phase === "working" ? "Generating Wireframe…" : "Generate Wireframe"}
        </button>
      </div>
    </div>
  );
}

app.jsx

import { useState } from "react";
import Sidebar from "./components/Sidebar.jsx";
import TopBar from "./components/TopBar.jsx";
import GeneratorForm from "./components/GeneratorForm.jsx";
import PreviewPanel from "./components/PreviewPanel.jsx";

const INITIAL_SESSIONS = [
  { id: 1, title: "Q2 sales overview", market: "DE", when: "Today" },
  { id: 2, title: "Retail footfall by store", market: "UK", when: "Today" },
  { id: 3, title: "Churn drivers — enterprise", market: "APAC", when: "Yesterday" },
  { id: 4, title: "Marketing spend vs pipeline", market: "US", when: "Yesterday" },
  { id: 5, title: "Inventory turns by category", market: "DE", when: "Previous 7 days" },
  { id: 6, title: "NPS trend, rolling 12m", market: "FR", when: "Previous 7 days" },
];

function generateDashboard() {
  return new Promise((resolve) => {
    setTimeout(() => resolve({ status: "done", pages: [] }), 1200);
  });
}

export default function App() {
  const [sessions, setSessions] = useState(INITIAL_SESSIONS);
  const [activeId, setActiveId] = useState(INITIAL_SESSIONS[0].id);
  const [query, setQuery] = useState("");

  const [persona, setPersona] = useState("Executive");
  const [tool, setTool] = useState("power_bi");
  const [file, setFile] = useState(null);
  const [instructions, setInstructions] = useState("");
  const [phase, setPhase] = useState("idle");
  const [result, setResult] = useState(null);
  const [error, setError] = useState("");

  const active = sessions.find((s) => s.id === activeId) || null;

  const handleNew = () => {
    const id = Date.now();
    setSessions((prev) => [
      { id, title: "Untitled dashboard", market: "DE", when: "Today" },
      ...prev,
    ]);
    setActiveId(id);
    setFile(null);
    setInstructions("");
    setPhase("idle");
    setResult(null);
    setError("");
  };

  const handleGenerate = async () => {
    setPhase("working");
    setError("");
    setResult(null);
    try {
      const res = await generateDashboard({ file, persona, tool, instructions });
      setResult(res);
      setPhase("done");
    } catch (e) {
      setError(e.message || "Something went wrong");
      setPhase("error");
    }
  };

  return (
    <div className="flex h-screen w-full bg-slate-50 text-slate-800 antialiased">
      {/* Sidebar */}
      <Sidebar
        sessions={sessions}
        activeId={activeId}
        query={query}
        onQuery={setQuery}
        onSelect={setActiveId}
        onNew={handleNew}
      />

      {/* Main Container */}
      <div className="flex min-w-0 flex-1 flex-col">
        <TopBar
          title={active ? active.title : "New dashboard"}
          market={(active && active.market) || "DE"}
        />

        {/* Scrollable Workspace */}
        <main className="flex-1 overflow-y-auto p-4 lg:p-6">
          <div className="mx-auto flex max-w-7xl flex-col gap-4">
            {/* Header */}
            <div className="flex items-center justify-between">
              <div>
                <span className="text-[11px] font-bold uppercase tracking-wider text-[#E60000]">
                  Studio Config
                </span>
                <h1 className="text-xl font-bold tracking-tight text-slate-900">
                  Dashboard Wireframe Generator
                </h1>
              </div>
            </div>

            {/* 2-Column Responsive Cards */}
            <div className="grid grid-cols-1 items-start gap-5 lg:grid-cols-12">
              {/* Left Column: Form Controls */}
              <div className="rounded-xl border border-slate-200/80 bg-white p-4 shadow-xs lg:col-span-5">
                <GeneratorForm
                  persona={persona} setPersona={setPersona}
                  tool={tool} setTool={setTool}
                  file={file} setFile={setFile}
                  instructions={instructions} setInstructions={setInstructions}
                  phase={phase} onGenerate={handleGenerate}
                />
              </div>

              {/* Right Column: Preview Canvas */}
              <div className="min-h-[460px] rounded-xl border border-slate-200/80 bg-white p-4 shadow-xs lg:col-span-7">
                <PreviewPanel
                  phase={phase}
                  result={result}
                  error={error}
                  tool={tool}
                  persona={persona}
                />
              </div>
            </div>
          </div>
        </main>
      </div>
    </div>
  );
}
